1. How Selenium C#, NUnit, and MSTest Work Together

When building browser automation in .NET, two distinct layers work side by side:

  • Selenium WebDriver (Selenium.WebDriver): Controls the browser (Chrome, Edge, Firefox) via Selenium Manager, which automatically downloads matching browser drivers without manual path setup.
  • Test Runner Framework (NUnit or MSTest): Discovers test methods, executes setup/teardown hooks, evaluates assertions, and generates TRX/XML reports for Azure DevOps or Jenkins pipelines.

4. Step-by-Step Code Implementation

Step A: Browser Initialization Factory (Drivers/DriverFactory.cs)

File: Drivers/DriverFactory.cs

Centralizing browser configuration makes it effortless to toggle headless execution inside CI/CD pipelines.

using OpenQA.Selenium;
using OpenQA.Selenium.Chrome;

namespace WebAutomation.Tests.Drivers;

public static class DriverFactory
{
    public static IWebDriver CreateChromeDriver(bool headless = false)
    {
        var options = new ChromeOptions();
        options.AddArgument("--window-size=1920,1080");
        if (headless)
        {
            options.AddArgument("--headless=new");
        }

        var driver = new ChromeDriver(options);
        // Keep implicit wait at 0 so explicit WebDriverWait remains predictable
        driver.Manage().Timeouts().ImplicitWait = TimeSpan.Zero;
        return driver;
    }
}

Step B: Page Object with Explicit Waits (Pages/LoginPage.cs)

File: Pages/LoginPage.cs

Notice how the Page Object works identically regardless of whether you run tests with NUnit or MSTest.

using OpenQA.Selenium;
using OpenQA.Selenium.Support.UI;
using SeleniumExtras.WaitHelpers;

namespace WebAutomation.Tests.Pages;

public class LoginPage
{
    private readonly IWebDriver _driver;
    private readonly WebDriverWait _wait;

    private readonly By _usernameInput = By.Id("username");
    private readonly By _passwordInput = By.Id("password");
    private readonly By _loginButton   = By.CssSelector("button[type='submit']");
    private readonly By _flashBanner   = By.Id("flash");

    public LoginPage(IWebDriver driver)
    {
        _driver = driver;
        _wait = new WebDriverWait(driver, TimeSpan.FromSeconds(10));
    }

    public void Open() =>
        _driver.Navigate().GoToUrl("https://the-internet.herokuapp.com/login");

    public void LoginAs(string username, string password)
    {
        _wait.Until(ExpectedConditions.ElementIsVisible(_usernameInput)).SendKeys(username);
        _driver.FindElement(_passwordInput).SendKeys(password);
        _wait.Until(ExpectedConditions.ElementToBeClickable(_loginButton)).Click();
    }

    public string GetBannerMessage() =>
        _wait.Until(ExpectedConditions.ElementIsVisible(_flashBanner)).Text;
}

Step C1: Writing the Test Suite in NUnit (Tests/LoginNUnitTests.cs)

File: Tests/LoginNUnitTests.cs 
using NUnit.Framework;
using OpenQA.Selenium;
using WebAutomation.Tests.Drivers;
using WebAutomation.Tests.Pages;

namespace WebAutomation.Tests.Tests;

[TestFixture]
public class LoginNUnitTests
{
    private IWebDriver _driver = null!;
    private LoginPage _loginPage = null!;

    [SetUp]
    public void SetUp()
    {
        _driver = DriverFactory.CreateChromeDriver(headless: false);
        _loginPage = new LoginPage(_driver);
    }

    [Test]
    public void ValidCredentials_ShouldDisplaySecureAreaBanner()
    {
        _loginPage.Open();
        _loginPage.LoginAs("tomsmith", "SuperSecretPassword!");

        StringAssert.Contains("You logged into a secure area!", _loginPage.GetBannerMessage());
    }

    [TestCase("wronguser", "SuperSecretPassword!")]
    [TestCase("tomsmith", "wrongpass")]
    public void InvalidCredentials_ShouldDisplayErrorBanner(string user, string pass)
    {
        _loginPage.Open();
        _loginPage.LoginAs(user, pass);

        StringAssert.Contains("invalid", _loginPage.GetBannerMessage());
    }

    [TearDown]
    public void TearDown()
    {
        _driver.Quit();
        _driver.Dispose();
    }
}

Step C2: Equivalent Test Suite in MSTest (Tests/LoginMsTestTests.cs)

File: Tests/LoginMsTestTests.cs — Side-by-Side Syntax Comparison

If your team standardizes on Visual Studio's built-in MSTest framework, only the attributes and assertions change:

using Microsoft.VisualStudio.TestTools.UnitTesting;
using OpenQA.Selenium;
using WebAutomation.Tests.Drivers;
using WebAutomation.Tests.Pages;

namespace WebAutomation.Tests.Tests;

[TestClass]
public class LoginMsTestTests
{
    private IWebDriver _driver = null!;
    private LoginPage _loginPage = null!;

    [TestInitialize]
    public void TestInitialize()
    {
        _driver = DriverFactory.CreateChromeDriver(headless: false);
        _loginPage = new LoginPage(_driver);
    }

    [TestMethod]
    public void ValidCredentials_ShouldDisplaySecureAreaBanner()
    {
        _loginPage.Open();
        _loginPage.LoginAs("tomsmith", "SuperSecretPassword!");

        Assert.IsTrue(_loginPage.GetBannerMessage().Contains("You logged into a secure area!"));
    }

    [DataTestMethod]
    [DataRow("wronguser", "SuperSecretPassword!")]
    [DataRow("tomsmith", "wrongpass")]
    public void InvalidCredentials_ShouldDisplayErrorBanner(string user, string pass)
    {
        _loginPage.Open();
        _loginPage.LoginAs(user, pass);

        Assert.IsTrue(_loginPage.GetBannerMessage().Contains("invalid"));
    }

    [TestCleanup]
    public void TestCleanup()
    {
        _driver.Quit();
        _driver.Dispose();
    }
}
